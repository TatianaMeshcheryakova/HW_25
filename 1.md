class Chocolate:
    def __init__(self, price, addition="", has_nuts=False, num_nuts=0):
        if price <= 0:
            raise ValueError("Цена должна быть больше нуля")
        self.price = price
        self.addition = addition
        self.num_nuts = num_nuts

    def show_my_chocolate(self):
        print(f"Шоколадный батончик, добавка: {self.addition}" if self.addition else "Обычный шоколадный батончик")

    def count_nuts(self):
        print(f"Количество орехов: {self.num_nuts}" if self.num_nuts > 0 else "Нет орехов")

    def __str__(self):
        return f"Шоколадный батончик, {self.price} рублей"

# Примеры использования
choco1 = Chocolate(100, "Карамель", True, 10)
choco2 = Chocolate(150, "Миндаль", True, 20)
choco3 = Chocolate(50)
choco4 = Chocolate(80, "Ваниль", True, 5)
choco5 = Chocolate(120, "Кокос", True, 15)
choco6 = Chocolate(60, has_nuts=True, num_nuts=3)

for choco in [choco1, choco2, choco3, choco4, choco5, choco6]:
    print(choco)
    choco.show_my_chocolate()
    choco.count_nuts()
